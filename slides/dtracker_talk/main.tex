\begin{column}{0.51\textwidth}
  \begin{mathcard}[Công thức Tucker]
    \centering
    \(\widehat{\mathcal X}_d
      =\mathcal W^{core}\times_{key}W^{key}\times_{loc}W^{loc}\)
  \end{mathcard}
  \vspace{2.5mm}
  {\small
  \[
    \widehat x_{d,uv}(t)
    =\sum_{i=1}^{d_k}\sum_{j=1}^{d_l}
      w^{core}_{ij}(t)\,W^{key}_{iu}W^{loc}_{jv}
  \]}
  \vspace{1mm}
  \begin{itemize}
    \item \(W^{key}_{iu}\) lớn: từ khóa \(u\) thuộc nhóm \(i\).
    \item \(W^{loc}_{jv}\) lớn: địa điểm \(v\) thuộc nhóm \(j\).
    \item \(w^{core}_{ij}(t)\): “kênh” hoạt động của cặp nhóm.
  \end{itemize}
  \begin{takeaway}[Con số đáng nhớ]
    Ví dụ \(d_k=3,d_l=2\): khoảng 748 tham số thay cho 41.600 ô trong cửa sổ.
  \end{takeaway}
\end{column}

\begin{frame}[shrink=8]{Viết điều đó bằng một phương trình}{Math 03 \textperiodcentered\ reaction--diffusion}
\begin{mathcard}[Hệ phương trình cho từng kênh ẩn]
  \centering
  \[
    \frac{dw_{ij}}{dt}
    =\underbrace{a_{ij}w_{ij}}_{\text{tự tăng/giảm}}
    +\underbrace{\sum_{j'=1}^{d_l}d_{ijj'}\bigl(w_{ij'}-w_{ij}\bigr)}_{\text{lực kéo từ các nhóm khác}}
  \]
\end{mathcard}
\vspace{2mm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
  {\small
  \begin{itemize}
    \item \(a_{ij}>0\): kênh có xu hướng tăng; \(a_{ij}<0\): giảm.
    \item \(d_{ijj'}\): độ mạnh lan từ nhóm \(j'\) sang \(j\).
    \item Dấu của \((w_{ij'}-w_{ij})\) quyết định chiều chảy.
  \end{itemize}}
\end{column}
\begin{column}{0.46\textwidth}
  \begin{intuition}[Một bước bằng số]
    \(w=1,\ w'=4,\ a=0.1,\ d=0.5\)\\[1mm]
    Tốc độ thay đổi \(=0.1(1)+0.5(4-1)=1.6\)\\[1mm]
    Bước Euler: \(w_{\text{next}}\approx1+1.6=2.6\)
  \end{intuition}
\end{column}
\end{columns}
\vfill
{\small\color{DTNavy!65}Đổi \(w'=0\), lực kéo đổi dấu: \(0.5(0-1)=-0.5\). Cùng một “ống nối”, hướng chảy khác.}

      $\mathcal W^{core}$ & $d_k\times d_l\times L_c$\\
      $W^{key}$ & $d_k\times k$, không âm\\
      $W^{loc}$ & $d_l\times l$, không âm
    \end{tabular}
  \end{mathcard}
  \vspace{3mm}
  {\small
  \[
    \widehat x_{d,uv}(t)
    =\sum_{i=1}^{d_k}\sum_{j=1}^{d_l}
    w^{core}_{ij}(t)W^{key}_{iu}W^{loc}_{jv}
  \]}

    \item \(d_k\ll k,\ d_l\ll l\): số nhóm ẩn nhỏ hơn số từ khóa/địa điểm.

    \item Hệ số không âm giữ chiều của động học, phục vụ diễn giải.
    \item Paper dùng Tucker cho trend; seasonal dùng PARAFAC.
  \end{itemize}
  \begin{caution}[Typo]
    Paper viết \(d_l\ll k\) ở một chỗ; theo kích thước và ngữ cảnh phải là \(d_l\ll l\).
  \end{caution}
\end{column}
\end{columns}

\begin{frame}{Backup 03: ALS, MDL và streaming}{Chi tiết cho Q\&A}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
  \begin{mathcard}[ModelEstimation]
    \small
    \[
      \min\ \bigl\|\mathcal X^c-\widehat{\mathcal X}_d
      -\widehat{\mathcal X}_s-\widehat{\mathcal X}_o\bigr\|_F^2
    \]
    \begin{enumerate}
      \item LM cho hệ reaction--diffusion.
      \item Cập nhật nhân cho \(W^{key},W^{loc}\).
      \item ALS + giả nghịch đảo cho seasonal.
      \item Sparsify residual cho outlier.
      \item Lặp đến hội tụ.
    \end{enumerate}
  \end{mathcard}
\end{column}
\begin{column}{0.47\textwidth}
  \begin{mathcard}[MDL]
    \small
    \[
      \langle\mathcal X;\mathcal F\rangle
      =\langle\mathcal F\rangle
      +\langle\mathcal X\mid\mathcal F\rangle
    \]
    {\footnotesize
    Mô tả mô hình + mô tả phần dư. Tăng rank làm mô hình đắt hơn nhưng có thể giảm phần dư.
    Algorithm 4 thử đổi một rank \(\pm1\).}
  \end{mathcard}
  \vspace{2mm}
  {\small
  \(\mathcal X^c=\{X_{t_c-L_c+1},\dots,X_{t_c}\}\).\\
  Chi phí mỗi bước không tăng theo tổng chiều dài stream; vẫn có spike khi rank update.}
\end{column}
\end{columns}
