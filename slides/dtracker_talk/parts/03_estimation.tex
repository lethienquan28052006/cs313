\begin{frame}{MDL trong một slide}{Minimum Description Length}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.53\textwidth}
  \small Nguyên lý: mô hình tốt nhất là mô hình giúp \textbf{nén dữ liệu ngắn nhất}
  (hiểu được nhiều quy luật $\Rightarrow$ nén được nhiều).
  \vspace{2mm}
  \formula{\langle\mathcal X;\mathcal F\rangle=
  \underbrace{\langle\mathcal F\rangle}_{\text{bit lưu mô hình}}+
  \underbrace{\langle\mathcal X\mid\mathcal F\rangle}_{\text{bit lưu sai số}}}
  \vspace{1mm}
  \begin{itemize}
    \item $\langle\mathcal F\rangle$: mỗi tham số khác 0 tốn khoảng 32 bit cộng chỉ số vị trí.
    \item $\langle\mathcal X\mid\mathcal F\rangle$: sai số càng nhỏ càng ít bit (mã hóa theo phân phối Gaussian).
  \end{itemize}
\end{column}
\begin{column}{0.44\textwidth}
\centering
\begin{tikzpicture}[x=1cm,y=1cm]
  \draw[DTNavy,line width=1pt] (-2.2,0.3) -- (2.2,-0.3);
  \fill[DTNavy] (0,0) -- (-.25,-.7) -- (.25,-.7) -- cycle;
  \node[dtaccent,minimum width=1.7cm,minimum height=.8cm,align=center,font=\scriptsize] at (-1.6,0.85) {mô hình\\to};
  \node[dtgold,minimum width=1.7cm,minimum height=.8cm,align=center,font=\scriptsize] at (1.6,0.35) {sai số\\nhỏ};
  \node[font=\scriptsize,text=DTMuted,align=center] at (0,-1.15) {mô hình to: khớp tốt nhưng tốn bit lưu\\mô hình nhỏ: rẻ nhưng sai số tốn bit};
\end{tikzpicture}
\vspace{2mm}
\begin{takeaway}[MDL dùng ở 3 chỗ]
  \small\numcircle{1} Điểm nào giữ làm \textbf{outlier}.\\[0.5mm]
  \numcircle{2} Chọn \textbf{$d_k,d_l,d_s$}.\\[0.5mm]
  \numcircle{3} Có nên \textbf{đổi mô hình} không.\\[1mm]
  $\Rightarrow$ Vì vậy D-Tracker \textbf{không cần hyperparameter}.
\end{takeaway}
\end{column}
\end{columns}
\vfill
\papersource{§4.2, Eq.~(9)(10). Công thức chi tiết: Appendix A6.}
\end{frame}

\begin{frame}{ModelUpdate: khi nào đổi mô hình, đổi rank}{Thích nghi}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\begin{mathcard}[Đổi mô hình]
  \small So sánh tổng cost:\\[1mm]
  \centering $\langle\mathcal X;\mathcal F\rangle$ \quad vs \quad $\langle\mathcal X;\mathcal F\cup\{\Theta'\}\rangle$\\[1mm]
  \raggedright Thêm $\Theta'$ mà rẻ hơn $\Rightarrow$ \textbf{sang giai đoạn mới}, dùng $\Theta'$ để mô hình hóa và dự báo.
\end{mathcard}
\vspace{2mm}
\begin{mathcard}[RankUpdate]
  \small Thử tăng/giảm \textbf{đúng một} trong $d_k,d_l,d_s$ một đơn vị (giả định cấu trúc đổi từ từ), giữ phương án có cost thấp nhất.
\end{mathcard}
\end{column}
