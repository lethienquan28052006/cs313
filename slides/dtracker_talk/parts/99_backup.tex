\begin{frame}[shrink=27]{Backup 01 --- Tensor/Tucker dimensions và lỗi in}{Technical backup \textperiodcentered\ Eq. (2)}
\formula{\widehat{\mathcal X}_d=\mathcal W^{core}\times_{key}W^{key}\times_{loc}W^{loc}}
\vspace{3mm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\begin{mathcard}[Kích thước]
  \begin{itemize}
    \item $k,l$: số keyword/location.
    $L_c$: độ dài window.
    \item $\mathcal W^{core}\in\mathbb R^{d_k\times d_l\times L_c}$.
    \item $W^{key}\in\mathbb R_+^{d_k\times k}$.
    $W^{loc}\in\mathbb R_+^{d_l\times l}$.
    \item Table 4: $(time,keyword,location)$; model equation: $(keyword,location,time)$.
  \end{itemize}
\end{mathcard}
\end{column}
\begin{column}{0.46\textwidth}
\begin{caution}[Likely typo trong paper]
  §3.3 viết $d_l\ll k$. Theo shape $W^{loc}\in\mathbb R_+^{d_l\times l}$ và ngữ cảnh nén location, điều phù hợp là $d_l\ll l$. Slide công bố đây là phán đoán theo kích thước, không “chỉnh lặng” paper.
\end{caution}
\vspace{3mm}
\begin{intuition}[Ý nghĩa nonnegative factors]
  Projection không đảo dấu latent relationships; giá trị cao biểu thị association mạnh hơn, không phải causal coefficient.
\end{intuition}
\end{column}
\end{columns}

\begin{frame}[shrink=44]{Backup 02 --- Bốn equations và initialization}{Technical backup \textperiodcentered\ model}
\small
\begin{align}
  \mathcal X^c&\approx\widehat{\mathcal X}_d+\widehat{\mathcal X}_s+\widehat{\mathcal X}_o,\\
  \widehat{\mathcal X}_d&=\mathcal W^{core}\times_{key}W^{key}\times_{loc}W^{loc},\\
  \frac{dw_{ij}}{dt}&=a_{ij}w_{ij}+\sum_{j'}d_{ijj'}(w_{ij'}-w_{ij}),\\
  \widehat{\mathcal X}_s&=S^{time}\times_{key}S^{key}\times_{loc}S^{loc}.
\end{align}
\vspace{2mm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{mathcard}[Parameter hierarchy]
  \begin{itemize}
    \item $\theta=\{A,\mathcal D\}$; $f_\theta(\theta,w_0,L)$ generates latent trajectory.
    \item $\theta_d=\{W^{key},W^{loc},\theta\}$.
    \item $\theta_s=\{S^{time},S^{key},S^{loc}\}$.
    \item $\Theta=\{\theta_d,\theta_s,\widehat{\mathcal X}_o\}$.
    $\mathcal F=\{\Theta_1,\ldots,\Theta_N\}$.
  \end{itemize}
\end{mathcard}
\end{column}
\begin{column}{0.48\textwidth}
\begin{intuition}[Initialization --- Appendix A.1]
  \begin{enumerate}
    \item STL tách trend / seasonal / residual.
    \item PARAFAC trên seasonal component để khởi tạo seasonal factors.
    \item Nonnegative Tucker Decomposition trên trend component để khởi tạo $W^{key},W^{loc}$.
  \end{enumerate}
\end{intuition}
\vspace{3mm}
\begin{takeaway}[Vai trò của sparse outliers]
  $\widehat{\mathcal X}_o$ chỉ giữ những entry bất thường mà sparsification theo MDL chấp nhận.
\end{takeaway}
\end{column}
\end{columns}
\vspace{2mm}
\papersource{§3.3, pp. 4--5; Appendix A.1, p. 10.}
\end{frame}

\begin{frame}[shrink=28]{Backup 03 --- Algorithms, MDL và complexity}{Technical backup \textperiodcentered\ optimization}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{mathcard}[ModelEstimation --- Algorithm 2]
  Initialize; repeat (1) reaction--diffusion, (2) trend factors, (3) seasonal factors, (4) outlier sparsification, đến convergence.
\end{mathcard}
\vspace{3mm}
\begin{intuition}[ModelUpdate + RankUpdate]
  Accept candidate khi MDL giảm; prose nói thử neighbor đổi một rank $\pm1$, re-estimate và giữ cost thấp hơn.
\end{intuition}
\vspace{3mm}
\begin{caution}[Printed Algorithm 4 issue]
  Candidate list lặp $(d_k,d_l+1,d_s)$ và không in $(d_k+1,d_l,d_s)$. Khi giải thích ý đồ $\pm1$, cần nói rõ duplication này.
\end{caution}
\end{column}
\begin{column}{0.48\textwidth}
\begin{mathcard}[MDL cost]
  \begin{itemize}
    \item $\langle\mathcal X;\mathcal F\rangle=\langle\mathcal F\rangle+\langle\mathcal X|\mathcal F\rangle$.
    \item Model cost dựa trên nonzero parameters; $c_F=32$ bits.
    \item Data cost: Gaussian residual negative log-likelihood ở đơn vị bits.
  \end{itemize}
\end{mathcard}
\vspace{3mm}
\begin{questioncard}
  \softlabel{LEMMA 1}\par
  ModelEstimation: $O(d_kd_l^2+kd_k+ld_l)$. Appendix A.3 giữ $\#iter$ ở intermediate bound rồi coi là hằng số nhỏ. Không dùng lemma này như complexity trọn gói của mọi seasonal/outlier/ODE/RankUpdate step.
\end{questioncard}
\end{column}
\end{columns}
\vspace{2mm}
\papersource{§4 pp. 5--7; Algorithm 2--4 và Appendix A.3, pp. 10--11.}
\end{frame}
